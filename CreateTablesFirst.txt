// Copy the following statements to create the tables and database

CREATE DATABASE IF NOT EXISTS csci_467;

CREATE TABLE User(user_id INT AUTO_INCREMENT PRIMARY KEY, associate_id VARCHAR(255) UNIQUE, Uusername VARCHAR(30), Upassword VARCHAR(30), address VARCHAR(30), commission VARCHAR(30)); 

CREATE TABLE quote(quote_id INT AUTO_INCREMENT PRIMARY KEY, event_date date, customer VARCHAR (255), associate VARCHAR (255), address VARCHAR(255), email VARCHAR(25), line_item VARCHAR(25), line_item_price VARCHAR(25), secret_note VARCHAR(255), amount INT, status VARCHAR(255));

CREATE TABLE orders(order_id VARCHAR(16) PRIMARY KEY, associate_id VARCHAR(255), quote_id INT, custid INT, amount VARCHAR(14), FOREIGN KEY (quote_id) REFERENCES quote(quote_id));
